CREATE VIEW contract_ledger_live_positions AS (
 SELECT account_id,
    full_path,
    commodity,
    round((balance_minor / (unit)::numeric), (- exponent)) AS balance,
    round((((accrued_num)::numeric / (accrued_den)::numeric) / (unit)::numeric), 7) AS accrued
   FROM ( SELECT a.id AS account_id,
            a.full_path,
            a.commodity,
            c.unit,
            c.exponent,
            (((COALESCE(p.balance, (0)::bigint))::numeric + COALESCE(( SELECT sum(m.amount) AS sum
                   FROM movements m
                  WHERE ((m.to_account_id = a.id) AND (m.value_time >= COALESCE(p.next_day, '0001-01-01 00:00:00'::timestamp without time zone)))), (0)::numeric)) - COALESCE(( SELECT sum(m.amount) AS sum
                   FROM movements m
                  WHERE ((m.from_account_id = a.id) AND (m.value_time >= COALESCE(p.next_day, '0001-01-01 00:00:00'::timestamp without time zone)))), (0)::numeric)) AS balance_minor,
            COALESCE(p.accrued_num, (0)::bigint) AS accrued_num,
            COALESCE(p.accrued_den, (1)::bigint) AS accrued_den
           FROM ((accounts a
             JOIN commodities c ON (((c.code)::text = (a.commodity)::text)))
             LEFT JOIN balances_live p ON (((p.account_id = a.id) AND (p.balance_date = ( SELECT max(q.balance_date) AS max
                   FROM balances_live q
                  WHERE (q.account_id = a.id))))))) l
)
