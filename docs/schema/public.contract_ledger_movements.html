Table Definition 
CREATE VIEW contract_ledger_movements AS (
 SELECT m.id,
    m.from_account_id,
    m.to_account_id,
    fa.full_path AS from_path,
    ta.full_path AS to_path,
    m.amount,
    m.code,
    m.value_time,
    m.knowledge_time,
    m.description
   FROM ((movements m
     JOIN accounts fa ON ((fa.id = m.from_account_id)))
     JOIN accounts ta ON ((ta.id = m.to_account_id)))
)
